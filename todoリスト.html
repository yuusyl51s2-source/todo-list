  function getRelativeDateStr(addDays) {

    const d = new Date();

    d.setDate(d.getDate() + addDays);

    return d.toISOString().split('T')[0];

  }

  function saveToStorage() {

    localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));

  }


  function loadFromStorage() {

    const data = localStorage.getItem(STORAGE_KEY);

    if (data) {

      try {

        const parsed = JSON.parse(data);

        if (parsed && parsed.workspaces) {

          appState = parsed;

        }

      } catch(e) {

        console.error("Storage load failed:", e);

      }

    }

  }


  function getCurrentWorkspace() {

    if (!appState.workspaces[appState.currentWorkspaceId]) {

      const keys = Object.keys(appState.workspaces);

      if (keys.length > 0) {

        appState.currentWorkspaceId = keys[0];

      } else {

        appState.workspaces['default_ws'] = {

          id: 'default_ws',

          name: 'メインのタスク表',

          categories: JSON.parse(JSON.stringify(DEFAULT_CATEGORIES)),

          tasks: []

        };

        appState.currentWorkspaceId = 'default_ws';

      }

    }

    return appState.workspaces[appState.currentWorkspaceId];

  }

  function requestNotificationPermission() {

    if (!("Notification" in window)) {

      alert("お使いのブラウザは通知機能をサポートしていません。");

      return;

    }


    Notification.requestPermission().then(permission => {

      updateNotifyButtonUI();

      if (permission === "granted") {

        new Notification("🌸 通知が許可されました", {

          body: "タスクの期限が3日前になると、ここでお知らせします！",

          icon: "https://fav.farm/🌸"

        });

        checkDueTasksForNotification();

      } else if (permission === "denied") {

        alert("ブラウザの設定で通知がブロックされています。ブラウザの設定から許可に変更してください。");

      }

    });

  }


  function updateNotifyButtonUI() {

    const btn = document.getElementById('notifyBtn');

    if (!("Notification" in window)) {

      btn.style.display = 'none';

      return;

    }


    if (Notification.permission === "granted") {

      btn.textContent = "🔔 通知オン";

      btn.classList.add("active");

    } else {

      btn.textContent = "🔔 通知を許可";

      btn.classList.remove("active");

    }

  }


  /* 期限3日前の判定・通知発行ロジック */

  function checkDueTasksForNotification() {

    if (!("Notification" in window) || Notification.permission !== "granted") return;


    const today = new Date();

    today.setHours(0, 0, 0, 0);


    let notifiedMap = {};

    try {

      notifiedMap = JSON.parse(localStorage.getItem(NOTIFIED_KEY) || '{}');

    } catch(e) {}


    // 全てのタスク表をスキャン

    Object.values(appState.workspaces).forEach(ws => {

      ws.tasks.forEach(task => {

        if (task.completed || !task.dueDate) return;


        const dueDate = new Date(task.dueDate + 'T00:00:00');

        const diffTime = dueDate.getTime() - today.getTime();

        const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));


        // 期限まで3日以内（3日, 2日, 1日, 当日, またはすでに期限超過）

        if (diffDays <= 3) {

          const notifyKey = `${task.id}_${task.dueDate}_${diffDays <= 0 ? 'due' : diffDays}`;


          if (!notifiedMap[notifyKey]) {

            let message = '';

            if (diffDays < 0) {

              message = `【期限超過】「${task.title}」の期限が過ぎています！`;

            } else if (diffDays === 0) {

              message = `【本日が期限】「${task.title}」は今日が期限です！`;

            } else {

              message = `【あと${diffDays}日】「${task.title}」の期限が近づいています（${task.dueDate}）`;

            }


            new Notification(`⏰ タスクリマインダー (${ws.name})`, {

              body: message,

              icon: "https://fav.farm/⏰"

            });


            notifiedMap[notifyKey] = true;

          }

        }

      });

    });


    localStorage.setItem(NOTIFIED_KEY, JSON.stringify(notifiedMap));

  }

  function renderFilterTabs() {

    allTab.onclick = () => setFilter('ALL');

      const tab = document.createElement('button');

      tab.className = `filter-tab ${activeFilter === cat.id ? 'active' : ''}`;

  function setFilter(catId) {

  function sortTasksByDueDate(tasks) {

    return tasks.slice().sort((a, b) => {

      if (a.completed !== b.completed) return a.completed ? 1 : -1;

      if (!a.dueDate && !b.dueDate) return a.id - b.id;

      if (!a.dueDate) return 1;

      if (!b.dueDate) return -1;

      if (a.dueDate !== b.dueDate) {

        return a.dueDate.localeCompare(b.dueDate);

      }

      return a.id - b.id;

    });

  }


  function renderTasks() {

    const ws = getCurrentWorkspace();

    const listEl = document.getElementById('taskList');

    listEl.innerHTML = '';


    let filtered = ws.tasks;

    if (activeFilter !== 'ALL') {

      filtered = ws.tasks.filter(t => t.categoryId === activeFilter);

    }


    if (filtered.length === 0) {

      listEl.innerHTML = `<div class="empty-state">🌸 タスクはありません</div>`;

      return;

    }


    const sorted = sortTasksByDueDate(filtered);

    const todayStr = new Date().toISOString().split('T')[0];


    sorted.forEach(task => {

      const cat = ws.categories.find(c => c.id === task.categoryId) || { name: '未分類', color: 'tea' };

      const colorObj = COLOR_OPTIONS.find(c => c.key === cat.color) || COLOR_OPTIONS[0];


      let dateHtml = '';

      if (task.dueDate) {

        const isUrgent = !task.completed && task.dueDate <= todayStr;

        dateHtml = `

          <span class="date-badge ${isUrgent ? 'urgent' : ''}">

            📅 ${task.dueDate} ${isUrgent ? '⚠️' : ''}

          </span>

        `;

      }


      let catOptionsHtml = ws.categories.map(c => `

        <option value="${c.id}" ${c.id === task.categoryId ? 'selected' : ''}>${c.name}</option>

      `).join('');


      const item = document.createElement('div');

      item.className = `task-item ${task.completed ? 'completed' : ''}`;

      item.innerHTML = `

        <input type="checkbox" class="task-checkbox" ${task.completed ? 'checked' : ''} onchange="toggleTask('${task.id}')">

        <div class="task-content">

          <div class="task-title">${escapeHtml(task.title)}</div>

          <div class="task-meta">

            <span class="cat-badge ${colorObj.class}">${cat.name}</span>

            ${dateHtml}

          </div>

        </div>

        <div class="task-item-actions">

          <select class="cat-select-mini" onchange="changeTaskCategory('${task.id}', this.value)">

            ${catOptionsHtml}

          </select>

          <button class="icon-btn" onclick="deleteTask('${task.id}')" title="削除">🗑️</button>

        </div>

      `;


      listEl.appendChild(item);

    });

  }

  function addTask() {

    const ws = getCurrentWorkspace();

    const titleInput = document.getElementById('taskTitle');

    const catSelect = document.getElementById('taskCategory');

    const dateInput = document.getElementById('taskDueDate');


    const title = titleInput.value.trim();

    if (!title) return;


    const newTask = {

      id: String(Date.now()),

      title: title,

      categoryId: catSelect.value || (ws.categories[0] ? ws.categories[0].id : ''),

      dueDate: dateInput.value,

      completed: false

    };


    ws.tasks.push(newTask);

    saveToStorage();


    titleInput.value = '';

    dateInput.value = '';


    renderFilterTabs();

    renderTasks();

    checkDueTasksForNotification();

  }


  function toggleTask(id) {

    const ws = getCurrentWorkspace();

    const task = ws.tasks.find(t => t.id === id);

    if (task) {

      task.completed = !task.completed;

      saveToStorage();

      renderTasks();

    }

  }


  function changeTaskCategory(taskId, newCatId) {

    const ws = getCurrentWorkspace();

    const task = ws.tasks.find(t => t.id === taskId);

    if (task) {

      task.categoryId = newCatId;

      saveToStorage();

      renderFilterTabs();

      renderTasks();

    }

  }


  function deleteTask(id) {

    const ws = getCurrentWorkspace();

    ws.tasks = ws.tasks.filter(t => t.id !== id);

    saveToStorage();

    renderFilterTabs();

    renderTasks();

  }

      `;

  function addNewCategoryField() {

    tempCategories.push({

      id: 'cat_' + Date.now(),

      name: '新しいカテゴリ',

      color: COLOR_OPTIONS[tempCategories.length % COLOR_OPTIONS.length].key

    });

    renderCatEditList();

  }


  function moveCategory(index, direction) {

    const targetIndex = index + direction;

    if (targetIndex < 0 || targetIndex >= tempCategories.length) return;

    const temp = tempCategories[index];

    tempCategories[index] = tempCategories[targetIndex];

    tempCategories[targetIndex] = temp;

    renderCatEditList();

  }


  function removeTempCategory(index) {

    if (tempCategories.length <= 1) {

      alert("カテゴリは最低1つ必要です。");

      return;

    }

    tempCategories.splice(index, 1);

    renderCatEditList();

  }


  function saveCategories() {

    const ws = getCurrentWorkspace();

    ws.categories = JSON.parse(JSON.stringify(tempCategories));


    const validIds = ws.categories.map(c => c.id);

    const defaultCatId = ws.categories[0].id;


    ws.tasks.forEach(task => {

      if (!validIds.includes(task.categoryId)) {

        task.categoryId = defaultCatId;

      }

    });


    saveToStorage();

    renderCategorySelect();

    renderFilterTabs();

    renderTasks();

    closeModal('catModal');

  }

  function escapeHtml(str) {

    return String(str)

      .replace(/&/g, '&amp;')

      .replace(/</g, '&lt;')

      .replace(/>/g, '&gt;')

      .replace(/"/g, '&quot;');

  }
